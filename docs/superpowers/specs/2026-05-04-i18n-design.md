**Date:** 2026-05-04
**Scope:** All 5 HTML pages — `index.html`, `2gether.html`, `fintonic.html`, `kimia.html`, `IoT.html`

### i18n.js structure

```javascript
(function () {
  var TRANSLATIONS = { en: {...}, es: {...} };

  function getKey(obj, path) {
    return path.split('.').reduce(function(o, k) { return o && o[k]; }, obj);
  }

  function applyLang(lang) {
    var t = TRANSLATIONS[lang];

    // Text content
    document.querySelectorAll('[data-i18n]').forEach(function(el) {
      var val = getKey(t, el.getAttribute('data-i18n'));
      if (val) el.textContent = val;
    });

    // Attribute translation (e.g. data-elements on typed span)
    document.querySelectorAll('[data-i18n-attr]').forEach(function(el) {
      var parts = el.getAttribute('data-i18n-attr').split(':');
      var val = getKey(t, parts[1]);
      if (val) el.setAttribute(parts[0], val);
    });

    // Re-init Typed.js if present on this page
    if (window.__typedInstance) {
      window.__typedInstance.destroy();
      window.__typedInstance = new Typed('.element', {
        strings: el.getAttribute('data-elements').split(', '),
        typeSpeed: 80, backSpeed: 50, backDelay: 1500, loop: true
      });
    }

    // Switcher active state
    document.querySelectorAll('[data-lang]').forEach(function(el) {
      el.classList.toggle('active', el.getAttribute('data-lang') === lang);
    });

    localStorage.setItem('preferred-lang', lang);
    document.documentElement.setAttribute('lang', lang);
  }

  function init() {
    var saved = localStorage.getItem('preferred-lang') || 'en';
    if (saved !== 'en') applyLang(saved);

    document.querySelectorAll('[data-lang]').forEach(function(el) {
      el.addEventListener('click', function(e) {
        e.preventDefault();
        applyLang(el.getAttribute('data-lang'));
      });
    });

    // Mark current lang active on load
    document.querySelectorAll('[data-lang]').forEach(function(el) {
      el.classList.toggle('active', el.getAttribute('data-lang') === saved);
    });
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', init);
  } else {
    init();
  }
}());
```

### typed.init.js change

Expose the instance globally so i18n.js can destroy/re-create it:
```javascript
// Before: new Typed('.element', {...})
// After:
window.__typedInstance = new Typed('.element', {...});
```

### Switcher HTML (added to every navbar, after `.nav-social` ul)

```html
<div class="lang-switcher ml-3">
  <a href="#" data-lang="en" class="lang-option active">EN</a>
  <span class="lang-sep">/</span>
  <a href="#" data-lang="es" class="lang-option">ES</a>
</div>
```
